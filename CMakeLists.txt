cmake_minimum_required(VERSION 3.21)

# MULTI-Module firmware, without the Arduino IDE
#
#   cmake --preset stm32f103cb
#   cmake --build --preset stm32f103cb
#
# The protocols and options are set in Multiprotocol/_Config.h (or
# _MyConfig.h), as for the Arduino IDE. The firmware is
# build/<preset>/multi-stm-<version>.bin.
#
# The MPM v2 boards (CC26xx) build the same protocol files, with the
# platform in platforms/cc26xx.

set(MULTI_CC26XX_BOARDS mpm-v2 launchpad xjt-lite)
set(MULTI_BOARD "stm32f103cb" CACHE STRING
  "Board: stm32f103cb, stm32f103c8, t18int, mpm-v2, launchpad, xjt-lite")
set_property(CACHE MULTI_BOARD PROPERTY STRINGS stm32f103cb stm32f103c8 t18int
  ${MULTI_CC26XX_BOARDS})

if(NOT CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_SOURCE_DIR}/cmake/arm-none-eabi.cmake")
endif()

project(Multiprotocol LANGUAGES C CXX ASM)

find_package(Python3 REQUIRED COMPONENTS Interpreter)

# The flags come from the board, whatever the build type
foreach(lang C CXX ASM)
  foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_${lang}_FLAGS_${cfg} "")
  endforeach()
endforeach()

set(MULTI_SKETCH_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Multiprotocol" CACHE PATH "The sketch (Multiprotocol.ino)")

if(MULTI_BOARD IN_LIST MULTI_CC26XX_BOARDS)
  add_subdirectory(platforms/cc26xx)
  return()
endif()

include(cmake/stm32f1.cmake)

# MULTI_OPTIONS: options over _Config.h (and _MyConfig.h), without editing
# it. A list of NAME (#define NAME), NAME=VALUE (#define NAME VALUE) and
# -NAME (#undef NAME), applied in that order after _Config.h; NAME=VALUE is
# also defined before it, for the settings _Config.h uses itself (NBR_BANKS).
# In -NAME, * matches the names in _Config.h: -*_CC2500_INO, no CC2500
# protocol.
set(MULTI_OPTIONS "" CACHE STRING
  "Options over _Config.h: NAME, NAME=VALUE, -NAME (-*_CC2500_INO: no CC2500 protocol)")

file(STRINGS "${MULTI_SKETCH_DIR}/_Config.h" _config_lines REGEX "#define")
set(MULTI_CONFIG_NAMES)
foreach(_line IN LISTS _config_lines)
  if(_line MATCHES "^[ \t/]*#define[ \t]+([A-Za-z0-9_]+)")
    list(APPEND MULTI_CONFIG_NAMES ${CMAKE_MATCH_1})
  endif()
endforeach()
list(REMOVE_DUPLICATES MULTI_CONFIG_NAMES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MULTI_SKETCH_DIR}/_Config.h")

# The header of the options (multi_options.h in dir) and the definitions
# before _Config.h, in <out_var>
function(multi_options_header dir out_var)
  set(header "// Generated by CMake: ${ARGN}\n")
  set(defines)
  foreach(opt IN LISTS ARGN)
    if(NOT opt MATCHES "^(-?)([A-Za-z0-9_*]+)(=(.*))?$")
      message(FATAL_ERROR "Option '${opt}': NAME, NAME=VALUE or -NAME")
    endif()
    set(undef "${CMAKE_MATCH_1}")
    set(name "${CMAKE_MATCH_2}")
    set(has_value "${CMAKE_MATCH_3}")
    set(value "${CMAKE_MATCH_4}")
    if(name MATCHES "[*]")
      if(NOT undef OR has_value)
        message(FATAL_ERROR "Option '${opt}': * only in -NAME")
      endif()
      string(REPLACE "*" "[A-Za-z0-9_]*" re "${name}")
      set(names ${MULTI_CONFIG_NAMES})
      list(FILTER names INCLUDE REGEX "^${re}$")
      if(NOT names)
        message(FATAL_ERROR "Option '${opt}': no such name in _Config.h")
      endif()
    else()
      set(names ${name})
    endif()
    foreach(n IN LISTS names)
      string(APPEND header "#undef ${n}\n")
      if(NOT undef)
        string(APPEND header "#define ${n} ${value}\n")
        if(has_value)
          list(APPEND defines "${n}=${value}")
        endif()
      endif()
    endforeach()
  endforeach()
  file(CONFIGURE OUTPUT "${dir}/multi_options.h" CONTENT "${header}" @ONLY)
  set(${out_var} ${defines} PARENT_SCOPE)
endfunction()

file(GLOB MULTI_SKETCH_FILES CONFIGURE_DEPENDS
  "${MULTI_SKETCH_DIR}/*.ino" "${MULTI_SKETCH_DIR}/*.h")

# A firmware: the sketch as one C++ file, as the Arduino builder makes it,
# in <dir>/sketch, with the options (MULTI_OPTIONS syntax) in ARGN; the
# firmware is <dir>/<target>.bin, with the signature, and a copy named
# <copy> (multi_firmware.py --copy)
function(multi_sketch target dir copy)
  set(sketch_cpp "${dir}/sketch/Multiprotocol.ino.cpp")
  set(preproc "${dir}/sketch/preproc.cpp")
  set(options_flags)
  if(NOT "${ARGN}" STREQUAL "")
    multi_options_header("${dir}/sketch" defines ${ARGN})
    list(TRANSFORM defines PREPEND -D)
    set(options_flags -DUSE_MULTI_OPTIONS ${defines} -iquote "${dir}/sketch")
    set(options_header "${dir}/sketch/multi_options.h")
  endif()
  add_custom_command(
    OUTPUT ${sketch_cpp} ${preproc}
    COMMAND Python3::Interpreter ${CMAKE_SOURCE_DIR}/cmake/ino2cpp.py
      --sketch ${MULTI_SKETCH_DIR} --main Multiprotocol.ino
      --out ${sketch_cpp} --preproc ${preproc}
      -- ${MULTI_PREPROCESS} ${options_flags} -iquote ${MULTI_SKETCH_DIR}
    DEPENDS ${MULTI_SKETCH_FILES} ${options_header} ${CMAKE_SOURCE_DIR}/cmake/ino2cpp.py
    COMMENT "Generating the sketch's Multiprotocol.ino.cpp (${target})"
    VERBATIM)

  maple_firmware(${target} ${sketch_cpp} ${MULTI_SKETCH_DIR})
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${dir}")
  target_compile_options(${target}_sketch PRIVATE ${options_flags})

  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${target}> ${dir}/${target}.bin
    COMMAND Python3::Interpreter ${CMAKE_SOURCE_DIR}/cmake/multi_firmware.py
      --elf $<TARGET_FILE:${target}> --size-tool ${CMAKE_SIZE}
      --max-size ${MULTI_MAX_SIZE} --max-data ${MULTI_MAX_DATA}
      --bin ${dir}/${target}.bin
      --preproc ${preproc}
      --header ${MULTI_SKETCH_DIR}/Multiprotocol.h
      --board ${MULTI_BOARD_NAME}
      --copy ${copy}
    VERBATIM)
endfunction()

multi_sketch(Multiprotocol "${CMAKE_CURRENT_BINARY_DIR}"
  "${CMAKE_CURRENT_BINARY_DIR}/multi-{type}-{version}.bin" ${MULTI_OPTIONS})

# cmake --build --preset <preset> --target flash
set(MULTI_FLASH_METHOD "" CACHE STRING
  "Upload method: auto, usb, serial (with the boot loader), serial-noboot (empty: the board's)")
set_property(CACHE MULTI_FLASH_METHOD PROPERTY STRINGS "" auto usb serial serial-noboot)
set(MULTI_FLASH_PORT "" CACHE STRING
  "Serial port of the module (empty: the only USB serial port; or MULTI_FLASH_PORT in the environment)")
find_program(STM32FLASH stm32flash)
find_program(DFU_UTIL dfu-util)
if(MULTI_FLASH_METHOD)
  set(_flash_method ${MULTI_FLASH_METHOD})
else()
  set(_flash_method ${MULTI_FLASH_DEFAULT})
endif()
add_custom_target(flash
  COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/cmake/multi_flash.py
    --bin $<TARGET_FILE_DIR:Multiprotocol>/Multiprotocol.bin
    --method ${_flash_method}
    --bootloader ${MULTI_BOOTLOADER} --erase-len ${MULTI_ERASE_LEN}
    --port "${MULTI_FLASH_PORT}"
    --stm32flash "$<$<BOOL:${STM32FLASH}>:${STM32FLASH}>"
    --dfu-util "$<$<BOOL:${DFU_UTIL}>:${DFU_UTIL}>"
  DEPENDS Multiprotocol
  USES_TERMINAL
  VERBATIM)

# cmake --build --preset <preset> --target start: a module plugged into the
# computer waits in its USB boot loader, its firmware starts (nothing written)
add_custom_target(start
  COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/cmake/multi_flash.py
    --method start
    --dfu-util "$<$<BOOL:${DFU_UTIL}>:${DFU_UTIL}>"
  USES_TERMINAL
  VERBATIM)
